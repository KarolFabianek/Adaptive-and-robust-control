function BTMS_MENU(cmd)
% BTMS_MENU() -- menu interaktywne
% BTMS_MENU('calosc') -- testy, 24 symulacje, 162 testy niepewnosci,
%                        porownanie Python, wykresy. Bez Simulink i NASA.
root=fileparts(mfilename('fullpath'));
cd(root); addpath(root); addpath(fullfile(root,'matlab'));
if nargin==0 || isempty(cmd)
    while true
        pomoc();
        wybor=strtrim(lower(input('\nBTMS | numer lub komenda (0 = koniec): ','s')));
        if any(strcmp(wybor,{'0','koniec','exit'}))
            fprintf('Wyjscie. Aby wrocic wpisz BTMS_MENU()\n'); return;
        end
        try
            wykonaj(wybor,root);
        catch ME
            fprintf(2,'BLAD BTMS: %s\n',ME.message);
            for k=1:numel(ME.stack)
                fprintf(2,'  -> %s, linia %d\n',ME.stack(k).name,ME.stack(k).line);
            end
            fprintf('Kontrola: BTMS_MENU(''status''); gotowe dane: BTMS_MENU(''gotowe'').\n');
        end
    end
else
    % Try-catch rowniez przy bezposrednim wywolaniu BTMS_MENU('...'):
    try
        wykonaj(strtrim(lower(char(cmd))),root);
    catch ME
        fprintf(2,'BLAD BTMS: %s\n',ME.message);
        for k=1:numel(ME.stack)
            fprintf(2,'  -> %s, linia %d\n',ME.stack(k).name,ME.stack(k).line);
        end
        fprintf('Popraw blad zanim uznasz, ze wyniki MATLAB sa gotowe.\n');
    end
end
end

function c = btms_cfg()
% Jawne PARAMETRY PROJEKTOWE, nie pomiary fabryczne ani kalibracja NASA.
c.dt=5; c.Tmax=7200;
c.ns=96; c.np=50; c.U_cell=3.6; c.Ah_cell=3.5; c.R_cell=0.028;
c.mass_cell=0.048; c.cp=900; c.Cfrac=0.76;
% Gcs: 1650 W/K = 4800 ogniw / 2,9 K/W na ogniwo 18650 (k_r = 0,43 W/(m K),
% Bhundiya i in., TFAWS 2018; R ~ 1/(4*pi*k*L)). Wczesniej 165 W/K (~29 K/W) - ok. 10x za malo.
% Tmax_cool to MOC [W] (limit odbioru ciepla przy u=1), nie temperatura.
c.Gcs=1650; c.Gamb=25; c.Gcool=470; c.Tmax_cool=6500;
c.Tcool=18; c.COP=2.6; c.pump_max=400;
c.soc0=.25; c.soc_target=.80; c.taper_soc=.70;
c.I_max=300; c.I_taper=130;
c.Tref=30; c.Tlim=40; c.thermal_start=36.5; c.thermal_full=39.5; c.I_min=110;
c.ff_gain=.70; c.Kp_const=.0725; c.Ki_const=47.5e-6;
% Gain Scheduling po rho = Ts - Tcool: Kp,Ki = PI * clip(rho0/rho, smin, smax).
c.gs_rho0=17; c.gs_smin=1; c.gs_smax=3;
c.R_scale=1; c.Gcs_scale=1; c.Gcool_scale=1;
c.k_cal=.025; c.k_cyc=5e-5;
% SOC postoju: 78% dla wszystkich strategii; 50% to osobna POLITYKA postoju (kolumna *_park050).
c.socpark_regular=.78; c.socpark_life=.50;
c.days_per_year=300; c.EFC_per_session=.55;
c.Qah=c.np*c.Ah_cell;
c.Rn=c.ns*c.R_cell/c.np;
c.Ctotal=c.ns*c.np*c.mass_cell*c.cp;
c.Cc=c.Cfrac*c.Ctotal; c.Cs=(1-c.Cfrac)*c.Ctotal;
c.En_kWh=c.ns*c.U_cell*c.Qah/1000;
end

function T = btms_compare_reference()
% Kontrola zgodnosci implementacji MATLAB z referencja Python.
% NIE jest to walidacja pomiarowa; obie implementacje uzywaja tego samego modelu.
root=fileparts(fileparts(mfilename('fullpath')));
refFile=fullfile(root,'results','sessions_summary.csv');
matFile=fullfile(root,'results','sessions_summary_matlab.csv');
if ~isfile(refFile), error('Brak danych referencyjnych results/sessions_summary.csv'); end
if ~isfile(matFile), error('Najpierw uruchom BTMS_MENU(''obliczenia''). Brak sesji MATLAB.'); end
R=readtable(refFile); M=readtable(matFile);
if height(R)~=24 || height(M)~=24
    error('Oczekiwano 24 wierszy w obu zestawach danych.');
end
rows=cell(24,6);j=0;
for i=1:height(M)
    ix=strcmp(string(R.strategy),string(M.strategy(i))) & abs(R.ambient_C-M.ambient_C(i))<1e-9;
    if sum(ix)~=1, error('Brak jednoznacznego przypadku: %s / %g C',string(M.strategy(i)),M.ambient_C(i)); end
    j=j+1;
    rows(j,:)={string(M.strategy(i)),M.ambient_C(i),...
        abs(M.time_min(i)-R.time_min(ix)),...
        abs(M.Tmax_C(i)-R.Tmax_C(ix)),...
        abs(M.energy_aux_kWh(i)-R.energy_aux_kWh(ix)),...
        abs(M.SOC_end(i)-R.SOC_end(ix))};
end
T=cell2table(rows,'VariableNames',{'strategy','ambient_C','dt_min','dTmax_C','dEaux_kWh','dSOC'});
fprintf('ZGODNOSC Z REFERENCJA PYTHON: max |dt|=%.8g min; |dTc|=%.8g K; |dE|=%.8g kWh; |dSOC|=%.8g\n',...
    max(T.dt_min),max(T.dTmax_C),max(T.dEaux_kWh),max(T.dSOC));
% Tolerancja obejmuje roznice zaokraglen implementacji, a nie blad pomiarowy.
assert(max(T.dt_min)<0.1 && max(T.dTmax_C)<0.05 && max(T.dEaux_kWh)<0.003 && max(T.dSOC)<1e-7,...
    'Roznica MATLAB / Python jest wieksza niz spodziewana. Sprawdz cfg i algorytm.');
writetable(T,fullfile(root,'results','matlab_python_comparison.csv'));
fprintf('OK: zapisano results/matlab_python_comparison.csv\n');
end

function [Kp,Ki] = btms_gains(c,strategy,Ts)
% Nastawy PI. Strategie PI*: stale nastawy (dostrojone w punkcie projektowym).
% Strategie GS*: GAIN SCHEDULING po rho = Ts - Tcool [K].
% Uzasadnienie: wzmocnienie obiektu dQcool/du = Gcool*(Ts-Tcool) zmienia sie
% z temperatura powierzchni (nie z SOC). PI dostrojono dla goracej powierzchni
% rho0 = 35 - 18 = 17 K; harmonogram utrzymuje iloczyn Kp*Gcool*rho staly,
% gdy rho maleje (kompensacja zmiennego wzmocnienia, Astrom i Wittenmark).
% Ograniczenie skali do [gs_smin, gs_smax] = [1, 3].
if any(strcmp(strategy,{'CC','PI','PI_FF','PI_SAFE'}))
    Kp=c.Kp_const; Ki=c.Ki_const;
else
    rho=max(Ts-c.Tcool, c.gs_rho0/c.gs_smax);
    k=min(c.gs_smax,max(c.gs_smin,c.gs_rho0/rho));
    Kp=c.Kp_const*k; Ki=c.Ki_const*k;
end
end

function T=btms_goal80()
% Warunki spelnienia celu SOH >= 80% po 10 latach (scenariusz modelowy).
% Szybkie ladowanie GS_LIFE odbywa sie przy Ta = 35 C. SREDNIA ROCZNA temperatura
% postoju i SOC postoju to osobne, jawne warunki (35 C srednio rocznie to rekord
% swiata - Dallol ok. 34,6 C; typowe goraca klimaty to 25-30 C).
% Funkcja niczego nie zaklada z gory o wyniku - wypisuje progi spelnienia celu.
root=fileparts(fileparts(mfilename('fullpath')));
out=fullfile(root,'results'); figDir=fullfile(root,'figures');
if ~isfolder(out),mkdir(out);end
if ~isfolder(figDir),mkdir(figDir);end
c=btms_cfg(); [session,tr]=btms_run_session(c,'GS_LIFE',35);
if session.status ~= "DONE",error('Charging did not reach the goal SOC.');end
meanTc=mean(tr(:,2)); Ipeak=max(tr(:,5));
parkTemps=[35 30 27 25 23 20]; parkSOCs=[.78 .5 .4];
T=table(); baseLife=[]; hotLife=[];
for temp=parkTemps
    for parkSoc=parkSOCs
        [l,s]=btms_life_goal(c,meanTc,Ipeak,35,temp,parkSoc,c.days_per_year);
        s.Tc_charge_max_C=session.Tmax_C;
        s.charge_time_min=session.time_min;
        s.temp_below_40=session.Tmax_C<40;
        T=[T;s]; %#ok<AGROW>
        if temp==35 && abs(parkSoc-.5)<1e-9,baseLife=l;end
        if temp==25 && abs(parkSoc-.5)<1e-9,hotLife=l;end
    end
end
writetable(T,fullfile(out,'goal80_feasibility_matlab.csv'));
assert(~isempty(baseLife) && ~isempty(hotLife));
% Niezmienniki (bez narzuconego werdyktu): chlodniejszy postoj starzeje mniej,
% a ladowanie miesci sie w limicie 40 C.
assert(hotLife.SOH_pct(end) > baseLife.SOH_pct(end));
assert(session.Tmax_C<40,'Charging pack exceeded the 40 C project temperature limit.');
figure('Color','w','Name','BTMS lifetime requirement 80%');
plot(baseLife.years,baseLife.SOH_pct,'LineWidth',2.1);hold on;
plot(hotLife.years,hotLife.SOH_pct,'LineWidth',2.4);
yline(80,'--','80% wymaganie','LineWidth',1.1);
grid on; xlabel('Lata eksploatacji'); ylabel('SOH (pojemnosciowy) [%]');
title('10 lat: modelowa wykonalnosc celu 80% SOH');
legend('Postoj srednio 35 C / 50% SOC (rekordowy klimat)', ...
       'Postoj srednio 25 C / 50% SOC (goracy klimat)','Granica 80%', ...
       'Location','southwest');
ax=gca;try, ax.Toolbar.Visible='off';catch,end
exportgraphics(gcf,fullfile(figDir,'matlab_goal80_storage.png'),'Resolution',170);
verdict={'NIE SPELNIA','SPELNIA'};
fprintf('\nCEL 80%% po 10 latach | szybkie ladowanie GS_LIFE przy Ta=35 C\n');
fprintf('GS_LIFE ladowanie: Tmax=%.2f C; czas=%.2f min\n',session.Tmax_C,session.time_min);
fprintf('Postoj srednio 35 C / 50%% SOC: %.2f%% SOH -> %s\n',baseLife.SOH_pct(end), ...
    verdict{1+double(baseLife.SOH_pct(end)>=80)});
fprintf('Postoj srednio 25 C / 50%% SOC: %.2f%% SOH -> %s\n',hotLife.SOH_pct(end), ...
    verdict{1+double(hotLife.SOH_pct(end)>=80)});
for parkSoc=parkSOCs
    ok=T.storage_temperature_C(T.meets_80 & abs(T.storage_SOC-parkSoc)<1e-9);
    if isempty(ok)
        fprintf('SOC postoju %2.0f%%: cel 80%% niespelniony w calym zakresie 20-35 C\n',100*parkSoc);
    else
        fprintf('SOC postoju %2.0f%%: cel 80%% spelniony dla sredniej temperatury postoju <= %g C\n',100*parkSoc,max(ok));
    end
end
fprintf('UWAGA: 35 C jako SREDNIA ROCZNA temperatura postoju to ekstremum klimatyczne\n');

function T = btms_life(c,meanTc,Ipeak,Tamb,park_soc)
% Scenariuszowy model SOH. Nie jest zidentyfikowany na NASA ani na pakiecie EV.
% Dkal monotonicznie kumuluje dodatnie przyrosty sqrt(t); amplituda sezonowa 7C.
Dcal=0;Dcyc=0; L=zeros(120,4);
cfac=1+.2*max(0,Ipeak/c.Qah-1);
for m=1:120
    seasonal=7*sin(2*pi*(mod(m-.5,12))/12);
    Tstore=Tamb+seasonal; Tcyc=meanTc+seasonal;
    droot=sqrt(m/12)-sqrt((m-1)/12);
    Dcal=Dcal+c.k_cal*droot*exp(.04*(Tstore-25))*exp(1.1*(park_soc-.5));
    Dcyc=Dcyc+(c.days_per_year/12)*c.EFC_per_session*c.k_cyc*exp(.03*(Tcyc-25))*cfac;
    L(m,:)=[m/12,100*max(0,1-Dcal-Dcyc),100*Dcal,100*Dcyc];
end
T=array2table(L,'VariableNames',{'years','SOH_pct','Dcal_pp','Dcyc_pp'});
end

function [life, summary] = btms_life_goal(c,meanTc,Ipeak,Tcharge,Tstorage,SOCstorage,annualSessions)
% 10-year scenario: Tcharge controls cycling, Tstorage controls CALENDAR
% aging independently. This is CONDITIONAL feasibility, not validation.
% No storage HVAC energy is accounted for in this simplified analysis.
% Requested target: SOH(10y) >= 80% (capacity retention).
Dcal=0; Dcyc=0; rows=zeros(120,4);
cfac=1 + 0.2*max(0,Ipeak/c.Qah-1);
for m=1:120
    seasonal=7*sin(2*pi*(mod(m-.5,12))/12);
    deltaRoot=sqrt(m/12)-sqrt((m-1)/12);
    Dcal=Dcal+c.k_cal*deltaRoot * exp(.04*(Tstorage+seasonal-25)) * ...
        exp(1.1*(SOCstorage-.5));
    Dcyc=Dcyc+(annualSessions/12)*c.EFC_per_session*c.k_cyc * ...
        exp(.03*(meanTc+seasonal-25))*cfac;
    rows(m,:)=[m/12,100*max(0,1-Dcal-Dcyc),100*Dcal,100*Dcyc];
end
life=array2table(rows,'VariableNames',{'years','SOH_pct','Dcal_pp','Dcyc_pp'});
summary=table(Tcharge,Tstorage,SOCstorage,annualSessions, ...
    life.SOH_pct(end),life.Dcal_pp(end),life.Dcyc_pp(end), ...
    life.SOH_pct(end)>=80, ...
    'VariableNames',{'charging_ambient_C','storage_temperature_C','storage_SOC', ...
    'sessions_per_year','SOH_10y_pct','Dcal_pp','Dcyc_pp','meets_80'});
end

function T=btms_local_jacobian(c)
% LOKALNA LINIARYZACJA W PUNKCIE ROWNOWAGI - NIE jest dowodem stabilnosci
% globalnej ani stabilnosci LPV.
% Punkt pracy = stan ustalony petli PI (integrator => Tc = Tref):
%   Ts* = Tref - Qgen/Gcs,  Qcool* = Qgen - Gamb*(Ts*-Ta),  u* = z* z bilansu.
% Nastawy GS liczone w Ts*; ich pochodna po Ts mnozy e = 0, wiec A ma te sama postac.
% Jesli punkt NIE istnieje (Ts* <= Tcool albo u* poza [0,1]), Tref jest
% nieosiagalne przy danym pradzie: petla musi sie nasycic, a analiza liniowa
% nie ma zastosowania (feasible=false, lambda=NaN).
ambients=[15 25 35]; currents=[120 210 300]; SOC=[.25 .52 .79]; strategies={'PI','GS'};
rows=cell(numel(ambients)*numel(currents)*numel(SOC)*numel(strategies),9); row=0;
for Ta=ambients
 for I=currents
  for soc=SOC
   for k=1:numel(strategies)
    st=strategies{k};
    Gcs=c.Gcs*c.Gcs_scale; G=c.Gcool*c.Gcool_scale;
    Tc=c.Tref;
    R=c.Rn*c.R_scale*min(1.4,max(.7,1+.2*(.5-soc)+.002*(25-Tc)));
    Qgen=I^2*R;
    Ts=Tc-Qgen/Gcs;
    [Kp,Ki]=btms_gains(c,st,Ts);
    Qcool=Qgen-c.Gamb*(Ts-Ta);
    if G*(Ts-c.Tcool) <= c.Tmax_cool   % galaz Gcool*u*(Ts-Tcool)
        Bu=G*(Ts-c.Tcool); u=Qcool/Bu; Bs=G*u;
    else                               % galaz limitu mocy Tmax_cool*u
        Bu=c.Tmax_cool; u=Qcool/Bu; Bs=0;
    end
    feasible = Ts>c.Tcool && u>=0 && u<=1;
    lam=NaN; tau=NaN;
    if feasible
        dR_dTc=-.002*c.Rn*c.R_scale;
        A=[1+c.dt*(I^2*dR_dTc-Gcs)/c.Cc, c.dt*Gcs/c.Cc, 0; ...
           c.dt*(Gcs-Bu*Kp)/c.Cs, 1-c.dt*(Gcs+c.Gamb+Bs)/c.Cs, -c.dt*Bu/c.Cs; ...
           c.dt*Ki, 0, 1];
        lam=max(abs(eig(A)));
        tau=-c.dt/log(lam);
    end
    row=row+1;
    rows(row,:)={string(st),soc,I,Ta,Ts,u,feasible,lam,tau};
   end
  end
 end
end
T=cell2table(rows,'VariableNames',{'strategy','SOC','I_A','Ta_C', ...
    'Ts_eq_C','u_eq','feasible','max_abs_lambda','tau_dominant_s'});
end

function T=btms_nasa_validate(filePath)
% NASA Ames B0005: oryginalne pomiary pojemnosci w cyklach rozladowania.
% 70/30 PODZIAL CHRONOLOGICZNY; modele Q=a+b*sqrt(N) i Q=a+b*N
% porownane z prognoza persystencji (ostatnia wartosc uczaca).
% NIE waliduje modelu cieplnego EV ani prognozy SOH 10-letniej.
root=fileparts(fileparts(mfilename('fullpath')));
if nargin<1 || isempty(filePath)
    filePath=fullfile(root,'nasa_data','B0005.mat');
end
if ~isfile(filePath),error('NASA: brak pliku %s',filePath);end
S=load(filePath);
if isfield(S,'B0005')
    r=S.B0005;
else
    vars=fieldnames(S);
    if isempty(vars),error('NASA: pusty plik MAT');end
    r=S.(vars{1});
end
if ~isstruct(r)||~isfield(r,'cycle')
    error('NASA: oczekiwano struktury B0005.cycle');
end
cycles=r.cycle;
capacities=zeros(numel(cycles),1); kvalid=0;
for k=1:numel(cycles)
    entry=cycles(k);
    if ~isfield(entry,'type')||~strcmpi(strtrim(char(entry.type)),'discharge')
        continue
    end
    if ~isfield(entry,'data')||~isstruct(entry.data)||~isfield(entry.data,'Capacity')
        continue
    end
    v=double(entry.data.Capacity);
    if isempty(v),continue;end
    v=v(1);
    if ~isfinite(v)||v<=0,continue;end
    kvalid=kvalid+1;capacities(kvalid)=v;
end
capacities=capacities(1:kvalid);
N=numel(capacities);
if N<15,error('NASA: za malo prawdziwych cykli rozladowania (N=%d)',N);end
cy=(1:N)';ntrain=floor(.7*N);
tr=1:ntrain; te=ntrain+1:N;
rmse=@(a,b) sqrt(mean((a-b).^2));
% Model 1 (pierwotny): Q=a+b*sqrt(N).
X=[ones(N,1),sqrt(cy)];
b=X(tr,:)\capacities(tr);
pred=X*b;
rmseTrain=rmse(capacities(tr),pred(tr));
rmseTest=rmse(capacities(te),pred(te));
% Model 2 (dodany po przegladzie): Q=a+b*N. Dla B0005 (test cykliczny)
% zanik jest w przyblizeniu liniowy; sqrt(N) jest zbyt optymistyczny.
XL=[ones(N,1),cy];
bL=XL(tr,:)\capacities(tr);
predLin=XL*bL;
rmseTrainLin=rmse(capacities(tr),predLin(tr));
rmseTestLin=rmse(capacities(te),predLin(te));
% Punkt odniesienia: persystencja (ostatnia wartosc ze zbioru uczacego).
rmseTestPers=rmse(capacities(te),capacities(ntrain));
T=table(N,ntrain,N-ntrain,rmseTrain,rmseTest,rmseTrainLin,rmseTestLin,rmseTestPers, ...
    capacities(1),capacities(end),...
    'VariableNames',{'n_cycles','train_n','test_n','RMSE_train_Ah','RMSE_test_Ah', ...
    'RMSE_train_lin_Ah','RMSE_test_lin_Ah','RMSE_test_persistence_Ah','C_first_Ah','C_last_Ah'});
if ~isfolder(fullfile(root,'results')),mkdir(fullfile(root,'results'));end
if ~isfolder(fullfile(root,'figures')),mkdir(fullfile(root,'figures'));end
writetable(T,fullfile(root,'results','nasa_validation_matlab.csv'));
values=table(cy,capacities,pred,predLin,'VariableNames', ...
    {'discharge_index','capacity_Ah','capacity_fit_Ah','capacity_fit_lin_Ah'});
writetable(values,fullfile(root,'results','nasa_B0005_discharge_matlab.csv'));
f=figure('Color','w','Name','NASA B0005 - rzeczywiste dane');
plot(cy,capacities,'.','MarkerSize',10,'Color',[.08,.45,.74]); hold on;
plot(cy,pred,'-','LineWidth',1.8,'Color',[.86,.35,.16]);
plot(cy,predLin,'-','LineWidth',1.8,'Color',[.10,.60,.35]);
xline(ntrain,'--','70% uczenie / 30% test');
grid on; xlabel('Numer cyklu rozladowania');ylabel('Pojemnosc [Ah]');
title('NASA B0005: pomiary vs modele sqrt(N) i liniowy');
legend('Pomiary NASA',sprintf('Q=a+b*sqrt(N), RMSE test %.3f Ah',rmseTest), ...
    sprintf('Q=a+b*N, RMSE test %.3f Ah',rmseTestLin),'Podzial chronologiczny','Location','best');
ax=gca; try, ax.Toolbar.Visible='off'; catch, end
exportgraphics(f,fullfile(root,'figures','nasa_B0005_matlab.png'),'Resolution',160);
fprintf('NASA: %d rzeczywistych pomiarow rozladowania.\n',N);
fprintf('RMSE_test: sqrt(N)=%.5f Ah | liniowy=%.5f Ah | persystencja=%.5f Ah\n', ...
    rmseTest,rmseTestLin,rmseTestPers);
end

function [S,tr] = btms_run_session(c,strategy,Ta)
% Powtarzalna sesja 25->80% SOC, jawny Euler dt=5 s.
x=[Ta;Ta;c.soc0;0]; maxN=round(c.Tmax/c.dt);
tr=zeros(maxN,13); n=0; t=0; qaux=0; qheat=0; rmse_sum=0; nsat=0; peak=x(1); maxres=0;
for k=1:maxN
    if x(3)>=c.soc_target-1e-10, break; end
    [xn,o]=btms_step(c,strategy,Ta,x);
    t=t+c.dt; qaux=qaux+o(5)*c.dt/3.6e6; qheat=qheat+o(3)*c.dt/3.6e6;
    rmse_sum=rmse_sum+o(8)^2; nsat=nsat+double(o(2)>=.999999);
    maxres=max(maxres,abs(o(10))); peak=max(peak,xn(1));
    n=n+1;
    tr(n,:)=[t/60,xn(1),xn(2),xn(3),o(1:6),o(7:9)];
    x=xn;
end
tr=tr(1:n,:);
if n==0, error('Pusta sesja: brak krokow symulacji'); end
S=struct('strategy',string(strategy),'ambient_C',Ta,'time_min',t/60,...
'Tmax_C',peak,'Tend_C',x(1),'energy_aux_kWh',qaux,'energy_heat_kWh',qheat,...
'SOC_end',x(3),'RMSE_error_K',sqrt(rmse_sum/size(tr,1)), ...
'sat_fraction',nsat/size(tr,1),'energy_balance_residual_J',maxres,...
'points',size(tr,1),'mean_Tc_C',mean(tr(:,2)),'Ipeak_A',max(tr(:,5)));
if x(3)>=c.soc_target-1e-9,S.status="DONE";else,S.status="TIMEOUT";end
end

function btms_sfcn_life80(block)
% Level-2 MATLAB S-function. R2025b: discrete states [Tc; Ts; SOC; z].
% Input: ambient temperature. Output: the 4 states. No interpreted Fcn block.
% Calls EXACTLY btms_step(), used by the ordinary MATLAB solver.
% Dialog parameter 1: initial pack temperature Ta0 [C] (= ambient at start).
% Poprawki po przegladzie: (1) warunek poczatkowy nie jest juz wpisany na
% sztywno (35 C), tylko pochodzi z parametru bloku; (2) po osiagnieciu

% zamiast "zamrozenia" temperatury do konca symulacji.
setup(block);
end

block.SimStateCompliance = 'DefaultSimState';
block.RegBlockMethod('PostPropagationSetup', @allocateState);
block.RegBlockMethod('InitializeConditions', @initialize);
block.RegBlockMethod('Outputs', @outputStates);

end

end

function initialize(block)
% Start: pakiet w temperaturze otoczenia Ta0, SOC = soc0 z btms_cfg, z = 0.
c = btms_cfg();
Ta0 = double(block.DialogPrm(1).Data);

end

end

% a temperatury dalej sie zmieniaja (stygniecie pakietu przy chlodzeniu).
[xnext, ~] = btms_step(c, 'GS_LIFE', Ta, x);
block.Dwork(1).Data = xnext;
end


================================================================================

================================================================================
function [xn,o] = btms_step(c,strategy,Ta,x)
% Jeden krok dt modelu termicznego + PI + anti-windup + feedforward.
% x = [Tc; Ts; SOC; z]. Brak modelu OCV/CC-CV; taper SOC jest przyblizeniem.
Tc=x(1); Ts=x(2); soc=x(3); z=x(4);
clamp=@(v,a,b) min(b,max(a,v));
f=clamp((soc-c.taper_soc)/(c.soc_target-c.taper_soc),0,1);
Iprof=c.I_max*(1-f)+c.I_taper*f;
if any(strcmp(strategy,{'PI_SAFE','GS_SAFE','GS_LIFE'}))
    fT=clamp((Tc-c.thermal_start)/(c.thermal_full-c.thermal_start),0,1);
    Itherm=c.I_max-(c.I_max-c.I_min)*fT;
else
    Itherm=c.I_max;
end
if strcmp(strategy,'GS_LIFE'), Icap=210; else, Icap=c.I_max; end
I=max(0,min([Iprof,Itherm,Icap,(c.soc_target-soc)*c.Qah*3600/c.dt]));
R=c.Rn*c.R_scale*clamp(1+.2*(.5-soc)+.002*(25-Tc),.7,1.4);
Qgen=I^2*R;
[Kp,Ki]=btms_gains(c,strategy,Ts);
if any(strcmp(strategy,{'PI_FF','GS_FF','PI_SAFE','GS_SAFE','GS_LIFE'}))
    FF=c.ff_gain*Qgen/c.Tmax_cool;
else
    FF=0;
end
e=Tc-c.Tref;
if strcmp(strategy,'CC')
    u=.20; raw=u; zn=z;
else
    raw=Kp*e+z+FF;
    u=clamp(raw,0,1);
    if ~((raw>=1 && e>0) || (raw<=0 && e<0))
        zn=z+Ki*c.dt*e;
    else
        zn=z;
    end
end
Qcool=max(0,min(c.Tmax_cool*u,c.Gcool*c.Gcool_scale*u*(Ts-c.Tcool)));
Qamb=c.Gamb*(Ts-Ta);
Qcs=c.Gcs*c.Gcs_scale*(Tc-Ts);
dTc=c.dt*(Qgen-Qcs)/c.Cc;
dTs=c.dt*(Qcs-Qamb-Qcool)/c.Cs;
Paux=c.pump_max*u^3+Qcool/c.COP;
xn=[Tc+dTc; Ts+dTs; min(c.soc_target,soc+I*c.dt/(3600*c.Qah)); zn];
o=[I,u,Qgen,Qcool,Paux,FF,zn,e,Itherm, ...
    c.Cc*dTc+c.Cs*dTs-c.dt*(Qgen-Qamb-Qcool)];
end

function T = btms_verify_simulink(simOut,Ta)
% Compare samples from independently executed Simulink model against
% the exact MATLAB session (same numerical transition equation).
% UWAGA: to test zgodnosci implementacji (ta sama funkcja btms_step),
% a nie niezalezna walidacja modelu.

root=fileparts(fileparts(mfilename('fullpath')));
raw=simOut.get('BTMS_states');
if ~isstruct(raw) || ~isfield(raw,'time') || ~isfield(raw,'signals')
    error('Simulink: To Workspace must use Structure With Time for BTMS_states.');
end
t=double(raw.time(:));
v=double(squeeze(raw.signals.values));
if size(v,2) ~= 4 && size(v,1) == 4, v=v.'; end
if size(v,1) ~= numel(t) || size(v,2) ~= 4
    error('Simulink returned dimensions %s, expected samples x 4.',mat2str(size(v)));
end
c=btms_cfg(); [s,tr]=btms_run_session(c,'GS_LIFE',Ta);
if s.status ~= "DONE", error('MATLAB reference run did not finish.'); end
wanted = round(tr(:,1)*60*1000);
[tf,loc] = ismember(wanted,round(t*1000));
if ~all(tf), error('Missing %d timestamps in Simulink output.',sum(~tf)); end
D=abs(v(loc,1:3)-tr(:,2:4));
T=table(max(D(:,1)),max(D(:,2)),max(D(:,3)), ...
    'VariableNames',{'max_error_Tc_K','max_error_Ts_K','max_error_SOC'});
T.pass = T.max_error_Tc_K < 1e-4 && T.max_error_Ts_K < 1e-4 && ...
    T.max_error_SOC < 1e-8;
out=fullfile(root,'results');if ~isfolder(out),mkdir(out);end
writetable(T,fullfile(out,'simulink_matlab_check.csv'));
disp(T);
if ~T.pass, error('Simulink != MATLAB: do not present model as verified.'); end
% Po zakonczeniu ladowania (SOC=80%) SOC ma byc stale, a temperatura
% rdzenia nie moze byc "zamrozona" (I=0 -> pakiet stygnie).
after = t > tr(end,1)*60 + 1e-6;
if nnz(after) >= 2
    assert(max(abs(v(after,3)-c.soc_target)) < 1e-9, 'SOC changed after end of charge');
    assert(v(find(after,1,'last'),1) < v(find(after,1,'first'),1), ...
        'Core temperature frozen after end of charge');
end
fprintf('PASS: Simulink states match the MATLAB GS_LIFE session.\n');
end

function main_btms()
% Punkt startowy (MATLAB; potrzebny jedynie MATLAB bazowy z table/writetable).
% UWAGA: nowa, jawna implementacja; wyniki NIE musza byc rowne starym slajdom.

out=fullfile(root,'results');fig=fullfile(root,'figures');

if ~exist(fig,'dir'),mkdir(fig);end
c=btms_cfg(); strategies={'CC','PI','GS','PI_FF','GS_FF','PI_SAFE','GS_SAFE','GS_LIFE'};
ambs=[15,25,35]; cases=struct([]); lrows=cell(24,6);idx=0;
for a=ambs
    for i=1:numel(strategies)
        st=strategies{i};idx=idx+1;
        [s,tr]=btms_run_session(c,st,a);cases=[cases;s]; %#ok<AGROW>
        % Ten sam SOC postoju (78%) dla wszystkich strategii - porownanie samego ladowania.
        L=btms_life(c,s.mean_Tc_C,s.Ipeak_A,a,c.socpark_regular);
        % Polityka postoju przy 50% SOC to osobna decyzja uzytkowania (nie cecha strategii).
        L50=btms_life(c,s.mean_Tc_C,s.Ipeak_A,a,c.socpark_life);
        lrows(idx,:)={string(st),a,L.SOH_pct(end),L.Dcal_pp(end),L.Dcyc_pp(end),L50.SOH_pct(end)};
        if a==35
            labels={'time_min','Tc_C','Ts_C','SOC','I_A','u','Qgen_W','Qcool_W','Paux_W','FF','integrator','e_K','I_therm_A'};
            writetable(array2table(tr,'VariableNames',labels),fullfile(out,['session_35C_' st '_matlab.csv']));
        end
    end
end
writetable(struct2table(cases),fullfile(out,'sessions_summary_matlab.csv'));
Life=cell2table(lrows,'VariableNames',{'strategy','ambient_C','SOH_10y_pct','Dcal_pp','Dcyc_pp', ...
    'SOH_10y_park050_pct'});
writetable(Life,fullfile(out,'life_summary_matlab.csv'));
% Model-based uncertainty stress test: does NOT constitute a formal robustness proof.
rr=cell(162,8);j=0;
for r=[.85,1,1.15]
for gc=[.8,1,1.2]
for gco=[.8,1,1.2]
for da=[-3,0,3]
    c2=c;c2.R_scale=r;c2.Gcs_scale=gc;c2.Gcool_scale=gco;
    for st={'PI_SAFE','GS_SAFE'}
        s=btms_run_session(c2,st{1},35+da);j=j+1;
        rr(j,:)={string(st{1}),r,gc,gco,da,s.Tmax_C,s.time_min,s.status};
    end
end
end
end
end
rr=cell2table(rr,'VariableNames',{'strategy','R_multiplier','Gcs_multiplier','Gcool_multiplier','Ta_offset_C','Tmax_C','time_min','status'});
writetable(rr,fullfile(out,'robustness_35C_matlab.csv'));
J=btms_local_jacobian(c);writetable(J,fullfile(out,'local_frozen_jacobian_matlab.csv'));
% NASA runs ONLY when requested: BTMS_MENU('nasa'). Never as part of basic experiments.
fprintf('NASA: opcjonalne; nie jest uruchamiane automatycznie.\n');
fprintf('BTMS completed: %d scenarios. Results in %s\n',numel(cases),out);
assert(all(arrayfun(@(v) v.status=="DONE",cases)),'Some cases did not finish');
assert(all(abs([cases.SOC_end]-c.soc_target)<1e-9),'SOC target mismatch');
assert(height(rr)==162,'Oczekiwano 162 testow niepewnosci');
assert(all(isfinite([cases.Tmax_C])) && all([cases.time_min]>0),'Niepoprawne wyniki sesji');
% Tozsamosc algebraiczna kroku Eulera: wykrywa blad w kodzie, nie waliduje fizyki.
assert(all([cases.energy_balance_residual_J]<1e-7),'Energy balance broken');
end

function quality_tests()
% Unit/invariant tests, not empirical validation.
c=btms_cfg();
assert(abs(c.Qah-175)<1e-10);
assert(abs(c.Rn-.05376)<1e-10);
assert(abs(c.En_kWh-60.48)<1e-10);
for st={'CC','PI','GS','PI_FF','GS_FF','PI_SAFE','GS_SAFE','GS_LIFE'}
 for Ta=[15,25,35]
  [s,T]=btms_run_session(c,st{1},Ta);
  assert(strcmp(s.status,'DONE') && abs(s.SOC_end-.8)<1e-9);
  assert(all(T(:,6)>=0 & T(:,6)<=1));
  assert(all(T(:,8)>=0)); % Qcool >= 0
  % Tozsamosc algebraiczna kroku Eulera - chroni spojnosc kodu, nie waliduje fizyki.
  assert(s.energy_balance_residual_J<1e-7);
  L=btms_life(c,s.mean_Tc_C,s.Ipeak_A,Ta,.5);
  assert(all(diff(L.SOH_pct)<=1e-9)); % SOH must NOT recover by seasonal temperature
 end
end
% Nominalnie przy 35 C wszystkie strategie z regulatorem trzymaja Tc < 40 C.
for st={'PI','GS','PI_SAFE','GS_SAFE','GS_LIFE'}
    s=btms_run_session(c,st{1},35); assert(s.Tmax_C<40,'%s >= 40 C',st{1});
end
% Model SOH: chlodniejszy postoj i nizszy SOC postoju starzeja mniej (niezmiennik).
[s,tr]=btms_run_session(c,'GS_LIFE',35);
[~,ordinary]=btms_life_goal(c,mean(tr(:,2)),max(tr(:,5)),35,35,.5,300);
[~,conditional]=btms_life_goal(c,mean(tr(:,2)),max(tr(:,5)),35,23,.4,300);
assert(s.Tmax_C<40 && conditional.SOH_10y_pct>ordinary.SOH_10y_pct);
% btms_life i btms_life_goal to ten sam model - musza dac ten sam wynik,
% gdy temperatura postoju = temperatura otoczenia ladowania.
L=btms_life(c,mean(tr(:,2)),max(tr(:,5)),35,.5);
assert(abs(L.SOH_pct(end)-ordinary.SOH_10y_pct)<1e-9,'btms_life != btms_life_goal');
% Zbieznosc numeryczna jawnego Eulera: dt=5 s vs dt=1 s (35 C).
c1=c; c1.dt=1;
for st={'PI','GS_SAFE','GS_LIFE'}
    a=btms_run_session(c,st{1},35); b=btms_run_session(c1,st{1},35);
    assert(abs(a.Tmax_C-b.Tmax_C)<0.1 && abs(a.time_min-b.time_min)<0.2, ...
        'Brak zbieznosci dt dla %s: %.3f vs %.3f C',st{1},a.Tmax_C,b.Tmax_C);
end
% Liniaryzacja w punkcie rownowagi: kazdy osiagalny punkt jest lokalnie stabilny,
% a przy 120 A rownowaga istnieje w kazdym klimacie.
J=btms_local_jacobian(c);
assert(all(J.max_abs_lambda(J.feasible)<1));
assert(all(J.feasible(J.I_A==120)));
% GS po Ts-Tcool: w punkcie projektowym rho0 rowny PI, nigdy ponizej nastaw PI.
[kp1,ki1]=btms_gains(c,'GS',c.Tcool+c.gs_rho0); [kp0,ki0]=btms_gains(c,'PI',c.Tcool+c.gs_rho0);
assert(kp1==kp0 && ki1==ki0);
for Ts=[10 20 30 40 50], assert(btms_gains(c,'GS',Ts)>=c.Kp_const); end
fprintf('Unit and invariance checks passed. Life scenario is model-only.\n');
end
